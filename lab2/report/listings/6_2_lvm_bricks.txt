# Остановка и удаление старого volume2 (на node1)
vlad@node1:~$ sudo gluster --mode=script volume stop volume2

vlad@node1:~$ sudo gluster --mode=script volume delete volume2

vlad@node1:~$ sudo rm -rf /var/brick2

# Установка LVM и создание thin-тома (node1, аналогично на node2)
vlad@node1:~$ sudo apt install -y lvm2 xfsprogs thin-provisioning-tools
[вывод установки пакетов опущен]
vlad@node1:~$ sudo pvcreate /dev/sdb

# Монтирование и запись данных на клиенте

vlad@client:~$ echo "before snapshot" | sudo tee /mnt/volume2/data.txt

vlad@client:~$ for i in $(seq 1 5); do echo $i | sudo tee /mnt/volume2/s$i.txt > /dev/null; done
